ALLOWED_COUNTRIES = ("KR", "US", "JP")


def register_user(email, age, country, username):
    if "@" not in email:
        return {"ok": False, "error": "invalid email"}
    if not (13 <= age <= 120):
        return {"ok": False, "error": "invalid age"}
    if country not in ALLOWED_COUNTRIES:
        return {"ok": False, "error": "invalid country"}
    if not (3 <= len(username) <= 20):
        return {"ok": False, "error": "invalid username"}
    return {"ok": True, "user": {"email": email, "age": age, "country": country, "username": username}}
