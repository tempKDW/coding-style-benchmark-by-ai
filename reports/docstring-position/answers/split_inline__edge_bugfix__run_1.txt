ALLOWED_CHANNELS = ("email", "sms", "push")


def send_notification(user, message, kind, priority):
    """Send a notification to a user."""
    # step 1: permission check — disabled users still get an audit log entry
    if not user.notifications_enabled:
        user.notification_log.append({"channel": None, "kind": kind, "ok": False})
        return {"ok": False, "error": "disabled", "channel": None, "body": None}
    # step 1b: kind whitelist (billing / marketing / security only)
    if kind not in ("billing", "marketing", "security"):
        return {"ok": False, "error": "invalid kind", "channel": None, "body": None}

    # step 2: template render — high priority gets an "[urgent] " prefix
    prefix = "[urgent] " if priority == "high" else ""
    body = prefix + message

    # step 3: channel select — high forces push; otherwise honour preferred_channel
    # if it is in ALLOWED_CHANNELS (case-insensitive), else fall back by priority
    preferred_lower = user.preferred_channel.lower() if user.preferred_channel else None
    if priority == "high":
        channel = "push"
    elif preferred_lower in ALLOWED_CHANNELS:
        channel = preferred_lower
    elif priority == "low":
        channel = "sms"
    else:
        channel = "email"

    # step 4: dispatch (mocked here; real impl would hit a provider SDK)
    sent = {"sent": True, "channel": channel, "body": body}

    # step 5: logging — append outcome for audit (success path)
    user.notification_log.append({"channel": channel, "kind": kind, "ok": sent["sent"]})
    return {"ok": True, "channel": channel, "body": body}
