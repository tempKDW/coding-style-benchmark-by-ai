def read_log(path):
    with open(path) as f:
        return f.readlines()


def parse(lines):
    events = []
    for line in lines:
        parts = line.strip().split(" ", 3)
        if len(parts) == 4:
            events.append({"timestamp": parts[0], "level": parts[1], "status": int(parts[2]), "message": parts[3]})
    return events


def filter_errors(events):
    return [e for e in events if e["status"] >= 500]


def compute_stats(events, error_events, recent_error_count):
    total = len(events)
    error_count = len(error_events)
    return {
        "total": total,
        "error_count": error_count,
        "error_rate": error_count / total if total else 0.0,
        "recent_error_count": recent_error_count,
    }


def format_report(stats):
    base = f"total={stats['total']} errors={stats['error_count']} rate={stats['error_rate']:.2%}"
    if "recent_error_count" in stats:
        base += f" recent_errors={stats['recent_error_count']}"
    return base


def process(path):
    events = parse(read_log(path))
    error_events = filter_errors(events)
    recent_error_count = sum(1 for e in error_events if e["timestamp"] >= "2024")
    return format_report(compute_stats(events, error_events, recent_error_count))
