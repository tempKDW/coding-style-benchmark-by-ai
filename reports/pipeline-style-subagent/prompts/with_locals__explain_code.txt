def read_log(path):
    with open(path) as f:
        return f.readlines()


def parse(lines):
    events = []
    for line in lines:
        parts = line.strip().split(" ", 3)
        if len(parts) == 4:
            events.append({"timestamp": parts[0], "level": parts[1], "status": int(parts[2]), "message": parts[3]})
    return events


def filter_errors(events):
    return [e for e in events if e["status"] >= 500]


def compute_stats(events, error_events):
    total = len(events)
    error_count = len(error_events)
    return {
        "total": total,
        "error_count": error_count,
        "error_rate": error_count / total if total else 0.0,
    }


def format_report(stats):
    return f"total={stats['total']} errors={stats['error_count']} rate={stats['error_rate']:.2%}"


def process(path):
    raw = read_log(path)
    events = parse(raw)
    error_events = filter_errors(events)
    stats = compute_stats(events, error_events)
    return format_report(stats)
