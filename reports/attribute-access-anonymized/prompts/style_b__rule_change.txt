def build_user_summary(user):
    summary = {}
    if hasattr(user, "display_name") and getattr(user, "display_name"):
        summary["display_name"] = getattr(user, "display_name")
    else:
        summary["display_name"] = "Anonymous"
    if hasattr(user, "email") and getattr(user, "email"):
        summary["email"] = getattr(user, "email")
    if hasattr(user, "avatar_url") and getattr(user, "avatar_url"):
        summary["avatar_url"] = getattr(user, "avatar_url")
    else:
        summary["avatar_url"] = "/default-avatar.png"
    if hasattr(user, "bio"):
        summary["bio"] = getattr(user, "bio")
    if hasattr(user, "location") and getattr(user, "location"):
        summary["location"] = getattr(user, "location")
    else:
        summary["location"] = "Unknown"
    if hasattr(user, "last_login") and getattr(user, "last_login"):
        summary["last_login"] = getattr(user, "last_login").isoformat()

    else:
